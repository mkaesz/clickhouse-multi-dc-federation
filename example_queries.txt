================================================================================
 FRA / MUC / HAM ClickHouse Geo Schema -- Example Queries
================================================================================

Tables involved:
  test_local          Tier 1 - physical data, ReplicatedMergeTree, per DC
  dist_test_regional   Tier 2 - intra-DC fan-out, per DC
  dist_test_global     Tier 3 - cross-DC fan-out, sharded by dc_name

Roles:
  app_writer  -- INSERT/SELECT on dist_test_regional and test_local only
  app_reader  -- SELECT on dist_test_global and dist_test_regional only


--------------------------------------------------------------------------------
 1. INSERTS
--------------------------------------------------------------------------------

-- 1.1 Normal application insert: write to the REGIONAL table, not the global
--     one. Run against the DC you're actually writing into (e.g. connect to
--     a MUC endpoint to insert MUC data). dc_name defaults automatically based
--     on which DC's test_local you're routed to via dist_test_regional.

INSERT INTO default.dist_test_regional (id, event_time, payload)
VALUES (1001, now(), 'sample payload from MUC');

-- 1.2 Batch insert, same pattern:

INSERT INTO default.dist_test_regional (id, event_time, payload)
VALUES
    (1002, now(), 'batch row 1'),
    (1003, now(), 'batch row 2'),
    (1004, now(), 'batch row 3');

-- 1.3 Insert directly into the local table (bypasses distributed routing --
--     only useful for single-shard DCs like ours today, or for backfills/
--     migrations where you want to write to a specific replica directly):

INSERT INTO default.test_local (id, event_time, payload, dc_name)
VALUES (1005, now(), 'direct local insert', 'FRA');

-- 1.4 Insert with explicit dc_name override (rare -- normally the column
--     default handles this; only needed if migrating/relabeling data):

INSERT INTO default.test_local (id, event_time, payload, dc_name)
VALUES (1006, now(), 'relabeled row', 'HAM');

-- 1.5 THIS WILL FAIL when run as a role-restricted user (app_writer /
--     app_reader have no INSERT grant on the global table). Connect as such
--     a user first:  clickhouse-client --user rbac_test_user
--     Running as the default admin user will succeed (it has full access).

INSERT INTO default.dist_test_global (id, event_time, payload, dc_name)
VALUES (9999, now(), 'should be rejected', 'FRA');
-- Expected (as rbac_test_user): Code: 497. DB::Exception: Not enough privileges...


--------------------------------------------------------------------------------
 2. SELECTS -- LOCAL / REGIONAL SCOPE
--------------------------------------------------------------------------------

-- 2.1 Read everything currently on this DC's local table:

SELECT * FROM default.test_local
ORDER BY event_time DESC
LIMIT 100;

-- 2.2 Read via the regional distributed table (same result as 2.1 today,
--     since each DC is a single shard -- but this is the query pattern
--     that keeps working unchanged if the DC grows to multiple shards):

SELECT * FROM default.dist_test_regional
WHERE event_time >= today()
ORDER BY event_time DESC
LIMIT 100;

-- 2.3 Basic count for this DC only:

SELECT count() FROM default.dist_test_regional;


--------------------------------------------------------------------------------
 3. SELECTS -- CROSS-DC (GLOBAL) SCOPE
--------------------------------------------------------------------------------

-- 3.1 Read across ALL three DCs:

SELECT dc_name, count() AS row_count
FROM default.dist_test_global
GROUP BY dc_name
ORDER BY dc_name;

-- 3.2 Read across all DCs, last hour only:

SELECT *
FROM default.dist_test_global
WHERE event_time >= now() - toIntervalHour(1)
ORDER BY event_time DESC
LIMIT 200;

-- 3.3 Scope to a SINGLE DC via the global table, with shard pruning enabled
--     (only queries MUC's shard, does not touch FRA or HAM):

SELECT *
FROM default.dist_test_global
WHERE dc_name = 'MUC'
ORDER BY event_time DESC
LIMIT 100
SETTINGS optimize_skip_unused_shards = 1;

-- 3.4 Scope to MULTIPLE specific DCs (IN works with pruning too --
--     queries only MUC + HAM shards, skips FRA):

SELECT
    dc_name,
    count() AS errors
FROM default.dist_test_global
WHERE dc_name IN ('MUC', 'HAM')
  AND event_time >= now() - toIntervalHour(1)
GROUP BY ALL
ORDER BY ALL ASC
SETTINGS optimize_skip_unused_shards = 1;

-- 3.5 Exclude a DC (e.g. everything except FRA):
--     NOTE: optimize_skip_unused_shards does NOT prune for != predicates --
--     all 3 shards are always scanned. Use IN() if you need pruning.

SELECT *
FROM default.dist_test_global
WHERE dc_name != 'FRA'
ORDER BY event_time DESC
LIMIT 100;

-- 3.6 Aggregate comparison across DCs:

SELECT
    dc_name,
    count() AS total_rows,
    min(event_time) AS earliest,
    max(event_time) AS latest
FROM default.dist_test_global
GROUP BY dc_name
ORDER BY dc_name;


--------------------------------------------------------------------------------
 4. VERIFYING SHARD PRUNING (diagnostic, not application queries)
--------------------------------------------------------------------------------

-- NOTE: EXPLAIN does NOT show pruning. It outputs a fixed 3-line logical plan
-- (output columns / blank / ReadFromRemote) regardless of WHERE clause or the
-- optimize_skip_unused_shards setting -- "3 rows in set" refers to those 3
-- plan-text rows, not data rows. Pruning is an execution-time optimisation and
-- is invisible to EXPLAIN.  Use system.query_log.read_rows instead.

-- NOTE: query_log is written asynchronously. Always run SYSTEM FLUSH LOGS
--       immediately before reading system.query_log to get current results.

-- 4.1 Single-DC pruned query: only HAM shard is contacted.
--     read_rows = rows returned FROM HAM shard (FRA and MUC not contacted):

SELECT * FROM default.dist_test_global
WHERE dc_name = 'HAM'
SETTINGS optimize_skip_unused_shards = 1;

SYSTEM FLUSH LOGS;

SELECT read_rows, query
FROM system.query_log
WHERE query LIKE '%dist_test_global%'
  AND type = 'QueryFinish'
  AND event_time >= now() - toIntervalMinute(2)
ORDER BY event_time DESC
LIMIT 1;

-- 4.2 IN() filter -- only matching shards are contacted (MUC skipped).
--     read_rows = rows returned from FRA shard + rows from HAM shard:

SELECT dc_name, count() FROM default.dist_test_global
WHERE dc_name IN ('FRA', 'HAM')
GROUP BY dc_name
SETTINGS optimize_skip_unused_shards = 1;

SYSTEM FLUSH LOGS;

SELECT read_rows, query
FROM system.query_log
WHERE query LIKE '%dist_test_global%'
  AND type = 'QueryFinish'
  AND event_time >= now() - toIntervalMinute(2)
ORDER BY event_time DESC
LIMIT 1;
-- read_rows here is 1 more than 4.1 (FRA contributes its matching rows too)
-- MUC's read_rows contribution = 0 because it was not contacted

-- 4.3 Without pruning -- all 3 shards always contacted.
--     Use the IN() variant so the result set differs from 4.1 (makes
--     the pruning vs. no-pruning read_rows comparison unambiguous):

SELECT dc_name, count() FROM default.dist_test_global
WHERE dc_name IN ('FRA', 'HAM')
GROUP BY dc_name;

SYSTEM FLUSH LOGS;

SELECT read_rows, query
FROM system.query_log
WHERE query LIKE '%dist_test_global%'
  AND type = 'QueryFinish'
  AND event_time >= now() - toIntervalMinute(2)
ORDER BY event_time DESC
LIMIT 1;
-- read_rows > 4.2 because MUC shard was also contacted (returned 0 matching rows
-- but still scanned its local table and returned to the initiator)

-- 4.4 Side-by-side comparison of recent queries (run 4.1–4.3 first):

SYSTEM FLUSH LOGS;

SELECT read_rows, left(query, 120) AS q
FROM system.query_log
WHERE query LIKE '%dist_test_global%dc_name%'
  AND type = 'QueryFinish'
  AND event_time >= now() - toIntervalMinute(5)
ORDER BY event_time DESC
LIMIT 5;


--------------------------------------------------------------------------------
 5. CLUSTER TOPOLOGY (diagnostic)
--------------------------------------------------------------------------------

-- 5.1 All configured clusters visible to this node (local + federated):
--     DISTINCT is required: the Replicated database engine registers the
--     'default' cluster independently, causing duplicate rows without it.

SELECT DISTINCT
    cluster,
    shard_num,
    replica_num,
    host_name,
    port,
    is_local
FROM system.clusters
ORDER BY cluster, shard_num, replica_num;

-- 5.2 Local intra-DC cluster only ('default' = operator-managed single shard):

SELECT DISTINCT
    shard_num,
    replica_num,
    host_name,
    port,
    is_local
FROM system.clusters
WHERE cluster = 'default'
ORDER BY shard_num, replica_num;

-- 5.3 Cross-DC federated cluster -- shows the 3 remote NodePort endpoints
--     injected by patch-federation.sh:

SELECT
    shard_num,
    replica_num,
    host_name,
    port,
    is_local,
    errors_count,
    estimated_recovery_time
FROM system.clusters
WHERE cluster = 'federated_dcs'
ORDER BY shard_num, replica_num;
-- Expected: 3 shards, shard 1=FRA, 2=MUC, 3=HAM  (ClickHouse is 1-based)
-- is_local=0 for all three -- NodePort IPs never match the local FQDN so
--   ClickHouse always treats them as remote, even for the local DC's own shard
-- errors_count > 0 means the remote is unreachable

-- 5.4 Active connection pool state per remote shard
--     (shows whether connections are established and healthy):

SELECT
    cluster,
    shard_num,
    host_name,
    port,
    errors_count,
    slowdowns_count,
    estimated_recovery_time
FROM system.clusters
WHERE cluster = 'federated_dcs'
ORDER BY shard_num;

-- 5.5 Macro substitutions active on this node (cluster/shard/replica identity):

SELECT macro, substitution
FROM system.macros
ORDER BY macro;
-- Note: the 'shard' and 'replica' macros are 0-based (0, 1, 2 ...) while
-- system.clusters.shard_num is 1-based. They refer to the same shards.

-- 5.6 Database engines -- shows what engine each database uses
--     (the 'default' database uses a Replicated engine managed by the operator):

SELECT
    name,
    engine,
    engine_full
FROM system.databases
WHERE name NOT IN ('system', 'information_schema', 'INFORMATION_SCHEMA')
ORDER BY name;

-- 5.7 ReplicatedMergeTree replica health for all local tables:

SELECT
    database,
    table,
    replica_name,
    is_leader,
    is_readonly,
    total_replicas,
    active_replicas,
    queue_size,
    inserts_in_queue,
    merges_in_queue
FROM system.replicas
ORDER BY database, table;
-- queue_size > 0 means replication lag; is_readonly=1 means Keeper is unreachable


--------------------------------------------------------------------------------
 6. NOTES
--------------------------------------------------------------------------------

- Always insert via dist_test_regional (Tier 2), never dist_test_global
  (Tier 3) -- this is enforced by RBAC (see 04_rbac/roles_and_grants.sql
  in the schema package), not just convention. See section 5 for topology
  queries that confirm which cluster/shard a node belongs to.

- optimize_skip_unused_shards only prunes reliably for equality / IN
  predicates directly on the sharding expression's input column (dc_name
  here). More complex WHERE clauses may not prune even if they're logically
  equivalent -- verify with system.query_log.read_rows (see section 4), not
  EXPLAIN (EXPLAIN does not reflect execution-time pruning).

- Any dc_name value outside 'FRA' / 'MUC' / 'HAM' falls back to shard 0
  (FRA) per the transform() default in the Tier 3 table definition.
